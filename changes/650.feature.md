Add a circuit constructor for the double bracket iteration given by https://arxiv.org/pdf/2206.11772v4. It can be used to rotate the state based on a Hamiltonian to give better results for VQEs. For example:

```python
import numpy as np
from qilisdk.analog import X, Y, Z
from qilisdk.backends import QiliSim
from qilisdk.cost_functions import ObservableCostFunction
from qilisdk.digital import CNOT, U3, Circuit, HardwareEfficientAnsatz
from qilisdk.functionals import DigitalPropagation, VariationalProgram
from qilisdk.optimizers import SciPyOptimizer
from qilisdk.readout import Readout

# Heisenberg chain
nqubits = 4
hamiltonian = sum(X(i) * X(i + 1) + Y(i) * Y(i + 1) + Z(i) * Z(i + 1) for i in range(nqubits - 1))

# Define the hardware-efficient ansatz for the VQE
ansatz = HardwareEfficientAnsatz(nqubits, layers=1, one_qubit_gate=U3, two_qubit_gate=CNOT)
ansatz.set_parameter_values(list(np.random.default_rng(0).uniform(-np.pi, np.pi, ansatz.nparameters)))

# Set up the backend and run the VQE to get the optimal parameters
backend = QiliSim()
vqe = VariationalProgram(DigitalPropagation(ansatz), SciPyOptimizer(method="L-BFGS-B"), ObservableCostFunction(hamiltonian))
ansatz.set_parameter_values(backend.execute(vqe, readout=Readout().with_state_tomography()).optimal_parameters)

# Improve the VQE state with two double-bracket steps, using it as the reference
diagonal = -sum(Z(i) for i in range(nqubits))
improved = Circuit.double_bracket_iteration(hamiltonian, [0.02, 0.02], diagonal=diagonal, reference=ansatz)

readout = Readout().with_expectation(observables=[hamiltonian])
print("VQE energy:     ", backend.execute(DigitalPropagation(ansatz), readout=readout).get_expectation_values()[0])
print("VQE + DBI energy:", backend.execute(DigitalPropagation(improved), readout=readout).get_expectation_values()[0])
# VQE energy:       -6.123105617453263
# VQE + DBI energy: -6.350472618513526
```
