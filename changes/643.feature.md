Added a period finding Circuit constructor, to help construct the quantum circuit needed for Shor's algorithm. Now the factor a number with QiliSDK using QiliSim for the quantum part:

```python
import math
import random
from fractions import Fraction

from qilisdk.backends import QiliSim
from qilisdk.digital import Circuit
from qilisdk.functionals import DigitalPropagation
from qilisdk.readout import Readout


def shor_via_qilisim(N: int) -> tuple[int, int]:
    while True:
        a = random.randrange(2, N)
        if math.gcd(a, N) > 1:
            return math.gcd(a, N), N // math.gcd(a, N)

        circuit = Circuit.period_finding(a, N)
        nprecision = circuit.nqubits - (N - 1).bit_length()
        samples = QiliSim().execute(DigitalPropagation(circuit), Readout().with_sampling(nshots=10)).get_samples()

        r = 1
        for bitstring in samples:
            m = int(bitstring[:nprecision], 2)
            r = math.lcm(r, Fraction(m, 2**nprecision).limit_denominator(N).denominator)

        if pow(a, r, N) == 1 and r % 2 == 0 and pow(a, r // 2, N) != N - 1:
            p = math.gcd(pow(a, r // 2, N) - 1, N)
            return p, N // p


print(shor_via_qilisim(21))  # returns (7, 3)
```
